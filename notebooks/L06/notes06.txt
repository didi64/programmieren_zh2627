def swap(items, i, j):
    item = items[i]
    items[i] = items[j]
    items[j] = item

def bubble_up_0(items):
    n = len(items)
    for i in range(n-1):
        if items[i] > items[i+1]:
            swap(items, i, i+1)

def bubble_up(items, k=0):
    n = len(items) - k
    for i in range(n-1):
        if items[i] > items[i+1]:
            swap(items, i, i+1)

def bubble_sort_0(items):
    n = len(items) - 1
    for i in range(n):
        bubble_up(items)

def bubble_sort(items):
    n = len(items) - 1
    for i in range(n):
        bubble_up(items, k=i)

def recursive_bubble_sort(items, k=0):
    print(f'{'    '*k}depth={k}, sorting {items[:len(items)-k]}')
    if len(items) - k < 2:
        return

    bubble_up(items, k)
    recursive_bubble_sort(items, k=k+1)

def delete_character(word, character):
    '''word: str
       character: str der Laenge 1
       entfernt aus word alle Vorkommmen von character
       und gibt das resulierende Wort zurueck
    '''
    result = ''
    for c in word:
        if c == character:
            continue
        result = result + c

    return result


def replace_char(word, character, new_character):
    '''ersetzt in word jedes Vorkommen von
       character durch new_character
    '''
    result = ''
    for c in word:
        if c == character:
            result = result + new_character
        else:
            result = result + c

    return result


def lstrip(word):
    '''entfernt allen Space vor und nach dem Wort
       und gibt dann das Wort zurueck
    '''
    SPACE = ' '
    n = len(word)

    for i in range(n):
        if word[i] != SPACE:
            break

    return word[i:]


def rstrip(word):
    '''entfernt allen Space vor und nach dem Wort
       und gibt dann das Wort zurueck
    '''
    SPACE = ' '
    n = len(word)

    for j in range(1, n+1):
        if word[-j] != SPACE:
            break

    return word[:n-j+1]


def strip(word):
    '''entfernt allen Space vor und nach dem Wort
       und gibt dann das Wort zurueck
    '''
    return lstrip(rstrip(word))


def find(s, sub, start=0, end=None):
    '''gib den ersten Index von Substring sub
       in s[start:end] zurueck oder -1
    '''
    s = s[start:end]
    if s == '' or sub == '':
        return -1              # early return

    n = len(s)
    m = len(sub)

    for i in range(n):
        if s[i:i+m] == sub:
            return start + i   # index, falls gefunden

    return -1                  # falls nicht gefunden


def get_chars_at_idxs(word, idxs):
    '''word: str
       idxs: tuple with indices
       returns the string formed of
       the characters word[i] for i in idxs

       get_positions('abcde', (2, 4)) -> 'ce'
    '''
    result = ''
    for i in idxs:
        result = result + word[i]

    return result


def get_occurrences(word, char):
    '''returns a tuple with all the positions
       where char occurs in word
    '''
    occurrences = ()
    n = len(word)

    for i in range(n):
        if (word[i] == char):
            occurrences = occurrences + (i,)

    return occurrences


def split(s, sep=','):
    '''Trenne den String s bei sep und
       gib ein Tuple mit den Teilen zurueck

       split('foo, bar, baz') -> ('foo', 'bar', 'baz')
       split('foo, bar, baz,') -> ('foo', 'bar', 'baz', '')
    '''
    items = ()
    occurrences = get_occurrences(s, sep) + (len(s),)

    start = 0
    for i in occurrences:
        item = s[start:i]
        item = strip(item)
        items = items + (item,)
        start = i+1

    return items
